def activate(t):
    a,p=approval(t);release,_,_,_=inputs(t);out=t/'release/approved-core.json'
    if out.exists():
        old=read(out)
        if old.get('parent_candidate_sha256')!=p['candidate_sha256'] or old.get('approval_sha256')!=sha(t/'approval.json'):raise ValueError('Existing activation has different identity')
        expected=dict(release,approved_for_training=True,parent_candidate_sha256=p['candidate_sha256'],approval_sha256=sha(t/'approval.json'))
        if old!=expected:raise ValueError('Existing activation differs from exact approved draft')
        return out
    release.update(approved_for_training=True,parent_candidate_sha256=p['candidate_sha256'],approval_sha256=sha(t/'approval.json'))
    with out.open('x',encoding='utf-8') as f:json.dump(release,f,indent=2,allow_nan=False)
    return out
